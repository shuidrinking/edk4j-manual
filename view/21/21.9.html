21.9、必须要将个别二进制内容的静态文件打包到jar时怎么办
方案一：pom中申明排除二进制文件，打包后再用jar命令将二进制文件合并进去
使用maven要将静态文件也打包到jar包时，如果静态文件中存在非文本内容的文件，如字体文件、证书文件等，可能会遇到异常 MalformedInputException。这通常是由于 Maven 在资源处理（filtering）过程中错误地尝试对二进制文件进行文本编码处理导致的。需要在静态文件申明中排除这些文件。 
<build>
	<resources>
		<resource>
				<directory>src/main/webapp</directory>
				<targetPath>static</targetPath>
				<filtering>true</filtering>
				<includes>
					<include>**/**</include>
				</includes>
				<excludes>
					<!-- 排除二进制文件，如：字体文件、证书文件、excel模板等 -->
					<exclude>**/*.woff</exclude>
					<exclude>**/*.ttf</exclude>
					<exclude>**/*.ico</exclude>
					<exclude>**/*.p12</exclude>
				</excludes>
			</resource>
	</resources>
</build>
	
#局部更新jar中文件的手法：将要更新的文件，按jar内的目录结构，把最外层文件夹复制到与jar平级，执行更新脚本
jar uvf xxxx.jar 单个文件在jar内的相对路径

#例：
#	当前所在位置
#		|__project.jar
#		|__static
#			|__xxxx
#				|__A.woff
#更新jar里woff文件，这样写：
jar -uvf project.jar static/xxxx/A.woff static/xxxx/A.woff
方案二：调整静态资源位置
将所有静态资源文件保存到特定目录：src/main/resources/META-INF/resources
使用maven插件处理静态资源 
<build>
	<plugins>
		<plugin>
			<groupId>org.apache.maven.plugins</groupId>
			<artifactId>maven-compiler-plugin</artifactId>
			<version>${maven-compiler-plugin.version}</version>
			<configuration>
				<source>${java.version}</source>
				<target>${java.version}</target>
				<encoding>${project.build.encoding}</encoding>
			</configuration>
		</plugin>
		<plugin>
			<groupId>org.apache.maven.plugins</groupId>
			<artifactId>maven-resources-plugin</artifactId>
			<version>3.3.1</version>
			<configuration>
				<encoding>${project.build.encoding}</encoding>
				<charset>${project.build.encoding}</charset>
			</configuration>
		</plugin>
	</plugins>
</build>
